# Medical AI Severity System — Quick Start
# =========================================

# ── STEP 1: Install dependencies ──────────────────────────────────────────
# Core pipeline
pip install langchain langchain-community langchain-groq
pip install xgboost shap scikit-learn joblib numpy pandas
pip install chromadb sentence-transformers
pip install fastapi uvicorn httpx python-dotenv pydantic

# ── STEP 2: Run smoke test (no API key needed) ─────────────────────────────
python medical_ai_e2e.py

# Expected output:
#   [Pipeline] Initializing components...
#   [RAG] Initialized with 12 documents
#   ✅ PASS  Domain correct (diabetes)
#   ✅ PASS  Severity score ≥ 60.0
#   ...
#   🎉 All tests passed!

# ── STEP 3: Add real LLM (free Groq API) ──────────────────────────────────
# 1. Get free key at: console.groq.com  (Llama 3 — 14,400 requests/day free)
# 2. Create .env file:
echo "GROQ_API_KEY=your_key_here" > .env

# 3. Run with real LLM synthesis:
python medical_ai_e2e.py --real

# ── STEP 4: Start the API server ──────────────────────────────────────────
python medical_ai_e2e.py --serve
# or
uvicorn medical_ai_e2e:app --reload

# API docs: http://localhost:8000/docs

# ── STEP 5: Test the API manually ─────────────────────────────────────────
curl -X POST http://localhost:8000/predict \
  -H "Content-Type: application/json" \
  -d '{
    "query_text": "Patient with glucose 210, BMI 34, frequent urination, age 52",
    "structured_input": {"glucose": 210, "bmi": 34, "age": 52},
    "patient_id": "demo_001"
  }'
