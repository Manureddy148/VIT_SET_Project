# ── STEP 1: Install dependencies ──────────────────────────────────────────
pip install -r requirements.txt
# Optional Chroma + embeddings (Layer 5). Omit if using MEDICAL_AI_SKIP_RAG=1:
pip install -r requirements-rag.txt

# ── STEP 2: Run tests ─────────────────────────────────────────────────────
# conftest sets MEDICAL_AI_SKIP_RAG=1 so CI stays light.
pytest

# ── STEP 3: Train and save real .joblib bundles (optional) ────────────────
python scripts/train_models.py

# ── STEP 4: Start the API (Layer 7) ───────────────────────────────────────
# EASIEST (opens browser, binds 0.0.0.0:8000 — keep window open!):
#   cd path\to\VIT_SET_Project
#   python launch_api.py
# Other options:
#   scripts\run_api.bat
#   .\scripts\run_api.ps1
# Diagnose: scripts\check_env.bat

# API docs: http://127.0.0.1:8000/docs

# ── STEP 5: Streamlit UI ─────────────────────────────────────────────────
# Terminal 1: API as above. Terminal 2:
streamlit run frontend/app.py
